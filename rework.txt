-- ==== UTILS ====
local function parseValue(text)
    text = text:gsub("/s","")
    local num = tonumber(text:match("(%d+%.?%d*)")) or 0
    local suffix = text:match("%d+%.?%d*(%a+)") or ""
    suffix = suffix:lower()
    if suffix=="k" then num=num*1e3 end
    if suffix=="m" then num=num*1e6 end
    if suffix=="b" then num=num*1e9 end
    return num
end

-- ==== BASE FINDER ====
local function setBaseToDeliveryHitbox()
    local plots = Workspace:FindFirstChild("Plots")
    if plots then
        for _,plot in ipairs(plots:GetChildren()) do
            local plotSign = plot:FindFirstChild("PlotSign")
            if plotSign then
                local surfaceGui = plotSign:FindFirstChild("SurfaceGui")
                if surfaceGui then
                    local frame = surfaceGui:FindFirstChild("Frame")
                    if frame then
                        local txt = frame:FindFirstChild("TextLabel")
                        if txt and txt.Text == plr.DisplayName.."'s Base" then
                            local hitbox = plot:FindFirstChild("DeliveryHitbox")
                            if hitbox and hitbox:IsA("BasePart") then
                                basePos = hitbox.Position
                                return
                            end
                        end
                    end
                end
            end
        end
    end
    basePos = root.Position + root.CFrame.LookVector*2
end
setBaseToDeliveryHitbox()

-- ==== ANTI-RAGDOLL ====
local function setAntiRagdoll()
    hum:ChangeState(Enum.HumanoidStateType.Running)
    hum.StateChanged:Connect(function(_,new)
        if new==Enum.HumanoidStateType.Ragdoll or new==Enum.HumanoidStateType.FallingDown then
            hum:ChangeState(Enum.HumanoidStateType.Running)
        end
    end)
end
setAntiRagdoll()

-- ==== PLAYER ESP ====
local function addESP(p)
    if p==plr then return end
    local function onChar(char)
        local hl = Instance.new("Highlight"); hl.Parent=char
        hl.FillTransparency=0.5; hl.FillColor=Color3.fromRGB(255,0,0)
        hl.OutlineTransparency=0; hl.OutlineColor=Color3.fromRGB(255,255,255)

        local bb = Instance.new("BillboardGui"); bb.Parent=char:WaitForChild("Head")
        bb.Adornee=char.Head; bb.Size=UDim2.new(0,150,0,40); bb.StudsOffset=Vector3.new(0,2,0); bb.AlwaysOnTop=true
        local txt = Instance.new("TextLabel",bb); txt.Size=UDim2.new(1,0,1,0); txt.BackgroundTransparency=1
        txt.Text=p.Name; txt.TextColor3=Color3.new(1,1,1); txt.TextSize=12
    end
    p.CharacterAdded:Connect(onChar)
    if p.Character then onChar(p.Character) end
end
for _,p in ipairs(Players:GetPlayers()) do addESP(p) end
Players.PlayerAdded:Connect(addESP)

-- ==== HIGHEST VALUE ESP ====
local function clearHighlights(tbl) for _,v in ipairs(tbl) do if v and v.Parent then v:Destroy() end end tbl={} end

local function updateHighestValueESP()
    clearHighlights(generationHighlights)
    local maxVal = -math.huge
    local plots = Workspace:FindFirstChild("Plots")
    if not plots then return end

    for _,plot in ipairs(plots:GetChildren()) do
        local plotSign = plot:FindFirstChild("PlotSign")
        local isLocal = false
        if plotSign then
            local txt = plotSign:FindFirstChild("SurfaceGui")?.Frame?.TextLabel
            if txt and txt.Text==plr.DisplayName.."'s Base" then isLocal=true end
        end
        if isLocal then continue end

        local podiums = plot:FindFirstChild("AnimalPodiums")
        if podiums then
            for _,pod in ipairs(podiums:GetChildren()) do
                local spawn = pod:FindFirstChild("Base")?.Spawn
                if spawn and spawn:IsA("BasePart") then
                    local att = spawn:FindFirstChild("Attachment")
                    if att then
                        local overhead = att:FindFirstChild("AnimalOverhead")
                        if overhead then
                            local gen = overhead:FindFirstChild("Generation")
                            if gen and gen:IsA("TextLabel") then
                                local val = parseValue(gen.Text)
                                if val>maxVal then
                                    maxVal=val
                                    maxPlotSign, maxSpawnPart, maxAnimalOverhead = plot:FindFirstChild("PlotSign"), spawn, overhead
                                    maxDeliveryHitbox, maxPurchases = plot:FindFirstChild("DeliveryHitbox"), plot:FindFirstChild("Purchases")
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    if maxSpawnPart and maxAnimalOverhead then
        -- spawn highlight
        local hl = Instance.new("Highlight"); hl.Parent=maxSpawnPart
        hl.FillTransparency=0.3; hl.FillColor=Color3.fromRGB(0,100,255); hl.OutlineTransparency=0; hl.OutlineColor=Color3.fromRGB(0,200,255)
        table.insert(generationHighlights,hl)

        -- full animal model highlight
        local model = maxSpawnPart.Parent.Parent.Parent
        if model and model:FindFirstChild("Humanoid") then
            local mhl = Instance.new("Highlight"); mhl.Parent=model
            mhl.FillTransparency=0.3; mhl.FillColor=Color3.fromRGB(0,255,0); mhl.OutlineTransparency=0; mhl.OutlineColor=Color3.fromRGB(255,255,0)
            table.insert(generationHighlights,mhl)
        end

        -- name billboard
        local disp = maxAnimalOverhead:FindFirstChild("DisplayName")
        if disp then
            local bb = Instance.new("BillboardGui"); bb.Parent=maxSpawnPart; bb.Adornee=maxSpawnPart
            bb.Size=UDim2.new(0,180,0,50); bb.StudsOffset=Vector3.new(0,6,0); bb.AlwaysOnTop=true
            local txt = Instance.new("TextLabel",bb); txt.Size=UDim2.new(1,0,1,0); txt.BackgroundTransparency=1
            txt.Text=disp.Text; txt.TextColor3=Color3.fromRGB(255,255,0); txt.Font=Enum.Font.Arcade; txt.TextSize=32
            txt.TextStrokeTransparency=0; txt.TextStrokeColor3=Color3.new(0,0,0)
            table.insert(generationHighlights,bb)
        end

        -- generation billboard
        local gen = maxAnimalOverhead:FindFirstChild("Generation")
        if gen then
            local bb = Instance.new("BillboardGui"); bb.Parent=maxSpawnPart; bb.Adornee=maxSpawnPart
            bb.Size=UDim2.new(0,180,0,50); bb.StudsOffset=Vector3.new(0,12,0); bb.AlwaysOnTop=true
            local txt = Instance.new("TextLabel",bb); txt.Size=UDim2.new(1,0,1,0); txt.BackgroundTransparency=1
            txt.Text=gen.Text; txt.TextColor3=Color3.fromRGB(0,255,0); txt.Font=Enum.Font.Arcade; txt.TextSize=32
            txt.TextStrokeTransparency=0; txt.TextStrokeColor3=Color3.new(0,0,0)
            table.insert(generationHighlights,bb)
        end
    end
end

-- ==== ESP LOCK (remaining time) ====
local function updateESPLock()
    clearHighlights(espLockHighlights)
    local plots = Workspace:FindFirstChild("Plots")
    if not plots then return end
    for _,plot in ipairs(plots:GetChildren()) do
        local purchases = plot:FindFirstChild("Purchases")
        if purchases and #purchases:GetChildren()>=2 then
            local main = purchases:GetChildren()[2]:FindFirstChild("Main")
            if main then
                local bb = main:FindFirstChild("BillboardGui")
                if bb then
                    local rem = bb:FindFirstChild("RemainingTime")
                    if rem and rem:IsA("TextLabel") then
                        local gui = Instance.new("BillboardGui"); gui.Parent=main; gui.Adornee=main
                        gui.Size=UDim2.new(0,180,0,60); gui.StudsOffset=Vector3.new(0,6,0); gui.AlwaysOnTop=true
                        local txt = Instance.new("TextLabel",gui); txt.Size=UDim2.new(1,0,1,0); txt.BackgroundTransparency=1
                        txt.Text=rem.Text; txt.TextColor3=Color3.new(1,1,1); txt.Font=Enum.Font.Arcade; txt.TextSize=36
                        txt.TextStrokeTransparency=0; txt.TextStrokeColor3=Color3.new(0,0,0)
                        table.insert(espLockHighlights,gui)
                    end
                end
            end
        end
    end
end

-- ==== FLY TO BEST ====
local function startFlyToBest()
    if not maxDeliveryHitbox or not maxDeliveryHitbox.Parent then return end
    if tick()-lastFlyAttempt < FLY_COOLDOWN then return end
    lastFlyAttempt = tick()

    if hum.WalkSpeed < 68 then
        local cloak = plr.Backpack:FindFirstChild("Invisibility Cloak") or chr:FindFirstChild("Invisibility Cloak")
        if cloak then hum:EquipTool(cloak); task.wait(0.6) end
        local remote = ReplicatedStorage:FindFirstChild("Packages")?.Net?.["RE/UseItem"]
        if remote then remote:FireServer() end
        task.wait(0.3)
    end

    local curY = root.Position.Y
    local target = Vector3.new(maxDeliveryHitbox.Position.X, curY, maxDeliveryHitbox.Position.Z)

    local dummy = Instance.new("Part"); dummy.Anchored=true; dummy.Transparency=1; dummy.CanCollide=false
    dummy.Position=target; dummy.CFrame=CFrame.new(target, target+root.CFrame.LookVector); dummy.Parent=Workspace

    local a0 = Instance.new("Attachment"); a0.Parent=root
    local a1 = Instance.new("Attachment"); a1.Parent=dummy
    local ap = Instance.new("AlignPosition"); ap.Attachment0=a0; ap.Attachment1=a1; ap.MaxForce=math.huge; ap.MaxVelocity=68; ap.Responsiveness=200; ap.Parent=root
    local ao = Instance.new("AlignOrientation"); ao.Attachment0=a0; ao.Attachment1=a1; ao.MaxTorque=math.huge; ao.Responsiveness=200; ao.Parent=root

    task.wait(3)
    ap:Destroy(); ao:Destroy(); a0:Destroy(); a1:Destroy(); dummy:Destroy()
end

-- ==== PUBLIC API ====
_G.CorruptionX = {
    basePos = function() return basePos end,
    flyToBest = startFlyToBest,
    toggleFloat = function(on)
        if on then
            _G.CorruptionX.floatPart = Instance.new("Part")
            local p = _G.CorruptionX.floatPart
            p.Size=Vector3.new(6,1,6); p.Anchored=true; p.Transparency=0.6; p.Material=Enum.Material.ForceField
            p.Color=Color3.fromRGB(100,150,255); p.CanCollide=true; p.Parent=Workspace
            task.spawn(function()
                while _G.CorruptionX.floatEnabled and p do
                    pcall(function() p.Position = root.Position - Vector3.new(0,3.2,0) end)
                    task.wait(0.05)
                end
            end)
        else
            if _G.CorruptionX.floatPart then _G.CorruptionX.floatPart:Destroy() _G.CorruptionX.floatPart=nil end
        end
        _G.CorruptionX.floatEnabled = on
    end,
    toggleSpeedBoost = function(on)
        _G.CorruptionX.speedBoost = on
        if on then
            task.spawn(function()
                while _G.CorruptionX.speedBoost and hum do
                    hum.WalkSpeed = 25
                    task.wait(0.001)
                end
            end)
        else
            if hum then hum.WalkSpeed = 16 end
        end
    end,
    toggleAutoSentry = function(on)
        _G.CorruptionX.autoSentry = on
        if on then
            task.spawn(function()
                while _G.CorruptionX.autoSentry and hum and root do
                    local bat = plr.Backpack:FindFirstChild("Bat") or chr:FindFirstChild("Bat")
                    if bat and not chr:FindFirstChild("Bat") then hum:EquipTool(bat); task.wait(0.5) end
                    for _,obj in ipairs(Workspace:GetDescendants()) do
                        if obj:IsA("BasePart") and obj.Name:find("Sentry") then
                            local dist = (obj.Position-root.Position).Magnitude
                            if dist<75 then
                                obj.CanCollide=false
                                for i=1,3 do
                                    obj.Position = root.Position + Vector3.new(0,0,2)
                                    if bat then bat:Activate() end
                                    task.wait(0.2)
                                end
                            end
                        end
                    end
                    task.wait(0.5)
                end
            end)
        end
    end,
    goToBase = function()
        if not basePos then return false end
        if _G.CorruptionX.isGoing then return true end
        _G.CorruptionX.isGoing = true

        -- STEP 1: Activate Falling State for 0.2s
        hum:ChangeState(Enum.HumanoidStateType.Freefall)
        task.wait(0.2)

        -- STEP 2: Wait 0.1s more (total 0.3s from start)
        task.wait(0.1)

        -- STEP 3: Start Tween + Reset State
        local curY = root.Position.Y
        local target = Vector3.new(basePos.X, curY, basePos.Z)
        hum:ChangeState(Enum.HumanoidStateType.Running) -- Reset state

        local dummy = Instance.new("Part"); dummy.Anchored=true; dummy.Transparency=1; dummy.CanCollide=false
        dummy.Position=target; dummy.CFrame=CFrame.new(target, target+root.CFrame.LookVector); dummy.Parent=Workspace

        local a0 = Instance.new("Attachment"); a0.Parent=root
        local a1 = Instance.new("Attachment"); a1.Parent=dummy
        local ap = Instance.new("AlignPosition"); ap.Attachment0=a0; ap.Attachment1=a1; ap.MaxForce=math.huge; ap.MaxVelocity=24; ap.Responsiveness=200; ap.Parent=root
        local ao = Instance.new("AlignOrientation"); ao.Attachment0=a0; ao.Attachment1=a1; ao.MaxTorque=math.huge; ao.Responsiveness=200; ao.Parent=root

        local origSpeed = hum.WalkSpeed; hum.WalkSpeed=30
        local start = tick()
        while _G.CorruptionX.isGoing and (root.Position-target).Magnitude>1 and (tick()-start)<30 do
            -- Medusa dodge
            for _,p in ipairs(Players:GetPlayers()) do
                if p~=plr and p.Character then
                    local tr = p.Character:FindFirstChild("HumanoidRootPart")
                    if tr and (tr.Position-root.Position).Magnitude<20 and p.Character:FindFirstChild("Medusa Head") then
                        root.Position = root.Position + Vector3.new(0,20,0)
                        curY = root.Position.Y; target = Vector3.new(basePos.X, curY, basePos.Z); dummy.Position=target
                        break
                    end
                end
            end
            task.wait(0.05)
        end

        -- cleanup
        ap:Destroy(); ao:Destroy(); a0:Destroy(); a1:Destroy(); dummy:Destroy()
        hum.WalkSpeed = origSpeed
        _G.CorruptionX.isGoing = false
        return true
    end,
    stopGoToBase = function()
        _G.CorruptionX.isGoing = false
        if hum then hum:ChangeState(Enum.HumanoidStateType.Running) end
    end,

    -- internal references (read-only for GUI)
    hum = hum, root = root,
    floatEnabled = false, speedBoost = false, autoSentry = false, isGoing = false
}

-- ==== MAIN LOOP ====
task.spawn(function()
    while true do
        updateHighestValueESP()
        updateESPLock()
        task.wait(3)
    end
end)

-- ==== KEYBINDS (optional – GUI can also call the API) ====
UserInputService.InputBegan:Connect(function(inp,gpe)
    if gpe then return end
    if inp.KeyCode==Enum.KeyCode.F then _G.CorruptionX.toggleFloat(not _G.CorruptionX.floatEnabled)
    elseif inp.KeyCode==Enum.KeyCode.G then _G.CorruptionX.goToBase()
    elseif inp.KeyCode==Enum.KeyCode.H then _G.CorruptionX.flyToBest()
    elseif inp.KeyCode==Enum.KeyCode.J then _G.CorruptionX.toggleSpeedBoost(not _G.CorruptionX.speedBoost)
    end
end)

-- ==== CLEANUP ====
game:GetService("CoreGui").ChildRemoved:Connect(function(child)
    if child.Name=="ThyroxideUI" then
        _G.CorruptionX.toggleFloat(false)
        _G.CorruptionX.toggleSpeedBoost(false)
        _G.CorruptionX.toggleAutoSentry(false)
        _G.CorruptionX.stopGoToBase()
        clearHighlights(generationHighlights)
        clearHighlights(espLockHighlights)
    end
end)

return _G.CorruptionX
